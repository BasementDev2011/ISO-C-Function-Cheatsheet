//|----------------------------------------------------------------------------
//| C23 Integer Type Definitions
//|----------------------------------------------------------------------------

// C23 also provides additional integer-related facilities through <stdint.h>.
// The exact-width types above are optional: an implementation defines them
// only when it has a corresponding integer type with exactly that width.
//
// int_leastN_t and uint_leastN_t are always provided for supported widths
// required by the implementation.
//
// int_fastN_t and uint_fastN_t provide integer types with at least N bits
// that are intended to be efficient for the implementation.
//
// intptr_t and uintptr_t are optional because an implementation may not have
// an integer type capable of holding a converted void pointer.
//
// intmax_t and uintmax_t are required and represent the widest signed and
// unsigned integer types available to the implementation.

//|----------------------------------------------------------------------------
//| Width Requirements
//|----------------------------------------------------------------------------

// N in the exact-width, least-width, and fast-width families refers to:
//
//     8
//     16
//     32
//     64
//
// Exact-width types have no padding bits and use two's complement for signed
// integer representation in modern ISO C versions where these types exist.
//
// The exact-width types are optional because ISO C does not require every
// implementation to provide an integer type of every specified width.

//|----------------------------------------------------------------------------
//| Common Usage
//|----------------------------------------------------------------------------

// Fixed-width integer:
//
//     int32_t x;
//
// Unsigned fixed-width integer:
//
//     uint32_t x;
//
// Maximum-width integer:
//
//     intmax_t x;
//
// Pointer-sized integer:
//
//     uintptr_t value;
//
// Portable integer constant:
//
//     UINT32_C(1000)
//
// Maximum integer value:
//
//     UINTMAX_MAX
