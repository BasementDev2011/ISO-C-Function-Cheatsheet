//|----------------------------------------------------------------------------
//| Program Termination (stdlib.h)
//|----------------------------------------------------------------------------

void          abort         (void);                                              /* Terminates the program abnormally. */
void          exit          (int status);                                        /* Terminates the program normally and performs normal termination processing. */
void          _Exit         (int status);                                        /* Terminates the program immediately without normal termination processing. */
int           atexit        (void (*function)(void));                            /* Registers a function to be called when the program terminates normally. */
int           at_quick_exit (void (*function)(void));                            /* Registers a function to be called when the program terminates through quick_exit. */
void          quick_exit    (int status);                                        /* Terminates the program while calling functions registered with at_quick_exit. */


//|----------------------------------------------------------------------------
//| Program Termination Constants (stdlib.h)
//|----------------------------------------------------------------------------

EXIT_FAILURE                                                                    /* Integer constant indicating unsuccessful program termination. */


//|----------------------------------------------------------------------------
//| Dynamic Memory Allocation (stdlib.h)
//|----------------------------------------------------------------------------

void         *malloc        (size_t size);                                       /* Allocates size bytes of uninitialized storage. */
void         *calloc        (size_t count, size_t size);                        /* Allocates storage for count objects of size bytes and initializes all allocated bytes to zero. */
void         *realloc       (void *pointer, size_t size);                       /* Changes the size of an existing memory allocation while preserving its existing contents. */
void          free          (void *pointer);                                     /* Releases memory previously allocated by malloc, calloc, or realloc. */
