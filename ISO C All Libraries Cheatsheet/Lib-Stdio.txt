//|----------------------------------------------------------------------------
//| File Types
//|----------------------------------------------------------------------------

FILE         /* Object type used to represent a stream. */
fpos_t       /* Object type used to represent a position within a file. */


//|----------------------------------------------------------------------------
//| End-of-File and Error Macros
//|----------------------------------------------------------------------------

EOF                                                               /* Negative integer constant returned by input functions to indicate end-of-file or an error. */
FOPEN_MAX                                                         /* Minimum number of files that can be open simultaneously. */
FILENAME_MAX                                                       /* Maximum size of a filename supported by the implementation. */
BUFSIZ                                                              /* Size of the buffer used by setbuf when a buffer is not otherwise specified. */
TMP_MAX                                                             /* Minimum number of unique filenames that can be generated by tmpnam. */
_IOFBF                                                              /* Flag used with setvbuf to request full buffering. */
_IOLBF                                                              /* Flag used with setvbuf to request line buffering. */
_IONBF                                                              /* Flag used with setvbuf to request no buffering. */
SEEK_SET                                                            /* Origin for fseek: beginning of the file. */
SEEK_CUR                                                            /* Origin for fseek: current file position. */
SEEK_END                                                            /* Origin for fseek: end of the file. */
EXIT_SUCCESS                                                        /* Value indicating successful program termination when passed to exit. */
EXIT_FAILURE                                                        /* Value indicating unsuccessful program termination when passed to exit. */


//|----------------------------------------------------------------------------
//| Standard Streams
//|----------------------------------------------------------------------------

stdin                                                               /* Standard input stream. */
stdout                                                              /* Standard output stream. */
stderr                                                              /* Standard error stream. */

//|----------------------------------------------------------------------------
//| File Operations
//|----------------------------------------------------------------------------

FILE         *fopen         (const char *restrict filename, const char *restrict mode); /* Opens a file and associates it with a stream. */
FILE         *freopen       (const char *restrict filename, const char *restrict mode, FILE *restrict stream); /* Closes the existing stream and opens filename using the specified mode. */
int           fclose        (FILE *stream);                                      /* Closes a stream and flushes any unwritten buffered output. */
int           fflush        (FILE *stream);                                      /* Writes buffered output to the associated file. */
int           remove        (const char *filename);                              /* Removes a file. */
int           rename        (const char *old_filename, const char *new_filename); /* Renames a file. */
FILE         *tmpfile       (void);                                              /* Creates a temporary file opened for update in binary mode. */
char         *tmpnam        (char *filename);                                    /* Generates a temporary filename that does not currently identify an existing file. */


//|----------------------------------------------------------------------------
//| File Positioning
//|----------------------------------------------------------------------------

int           fseek         (FILE *stream, long int offset, int origin);         /* Changes the file position indicator of a stream. */
long int      ftell         (FILE *stream);                                      /* Returns the current file position indicator. */
void          rewind        (FILE *stream);                                      /* Sets the file position indicator to the beginning of the stream and clears its error indicator. */
int           fgetpos       (FILE *restrict stream, fpos_t *restrict position);  /* Stores the current file position indicator in position. */
int           fsetpos       (FILE *stream, const fpos_t *position);              /* Sets the file position indicator to the position stored in position. */


//|----------------------------------------------------------------------------
//| Binary Input and Output
//|----------------------------------------------------------------------------

size_t        fread         (void *restrict destination, size_t size, size_t count, FILE *restrict stream); /* Reads count objects of size bytes from a stream into destination. */
size_t        fwrite        (const void *restrict source, size_t size, size_t count, FILE *restrict stream); /* Writes count objects of size bytes from source to a stream. */


//|----------------------------------------------------------------------------
//| Character Input
//|----------------------------------------------------------------------------

int           fgetc         (FILE *stream);                                      /* Reads the next character from stream and returns it as an unsigned char converted to int, or EOF. */
int           getc          (FILE *stream);                                      /* Reads the next character from stream and returns it as an unsigned char converted to int, or EOF. */
int           getchar       (void);                                              /* Reads the next character from stdin and returns it as an unsigned char converted to int, or EOF. */


//|----------------------------------------------------------------------------
//| Character Output
//|----------------------------------------------------------------------------

int           fputc         (int character, FILE *stream);                       /* Writes character to stream. */
int           putc          (int character, FILE *stream);                       /* Writes character to stream. */
int           putchar       (int character);                                     /* Writes character to stdout. */


//|----------------------------------------------------------------------------
//| String Input
//|----------------------------------------------------------------------------

char         *fgets         (char *restrict str, int count, FILE *restrict stream); /* Reads at most count - 1 characters from stream into str and terminates the string with a null character. */


//|----------------------------------------------------------------------------
//| String Output
//|----------------------------------------------------------------------------

int           fputs         (const char *restrict str, FILE *restrict stream);   /* Writes the string str to stream without automatically adding a newline. */
int           puts          (const char *str);                                   /* Writes str to stdout followed by a newline. */


//|----------------------------------------------------------------------------
//| Formatted Output
//|----------------------------------------------------------------------------

int           sprintf       (char *restrict str, const char *restrict format, ...); /* Writes formatted output to str without checking the destination size. */
int           snprintf      (char *restrict str, size_t size, const char *restrict format, ...); /* Writes formatted output to str while limiting the number of bytes written. */


//|----------------------------------------------------------------------------
//| Formatted Input
//|----------------------------------------------------------------------------

int           scanf         (const char *restrict format, ...);                 /* Reads formatted input from stdin. */
int           fscanf        (FILE *restrict stream, const char *restrict format, ...); /* Reads formatted input from stream. */
int           sscanf        (const char *restrict str, const char *restrict format, ...); /* Reads formatted input from a string. */


//|----------------------------------------------------------------------------
//| Variable-Argument Formatted Output
//|----------------------------------------------------------------------------

int           vprintf       (const char *restrict format, va_list arguments);    /* Writes formatted output to stdout using a va_list. */

//|----------------------------------------------------------------------------

//|----------------------------------------------------------------------------
//| Formatted I/O Conversion Specifications
//|----------------------------------------------------------------------------
//
// printf, fprintf, sprintf, and snprintf support these conversion specifiers:
//
// %d          Signed decimal integer.
// %i          Signed decimal integer.
// %o          Unsigned octal integer.
// %u          Unsigned decimal integer.
// %x          Unsigned hexadecimal integer using lowercase letters.
// %X          Unsigned hexadecimal integer using uppercase letters.
// %f          Floating-point value.
// %F          Floating-point value.
// %e          Floating-point value in exponent notation using e.
// %E          Floating-point value in exponent notation using E.
// %g          Floating-point value using the shorter of f or e.
// %G          Floating-point value using the shorter of F or E.
// %a          Floating-point value in hexadecimal notation using p.
// %A          Floating-point value in hexadecimal notation using P.
// %c          Character.
// %s          Null-terminated string.
// %p          Pointer.
// %n          Stores the number of characters written so far.
// %%          Literal percent character.


//|----------------------------------------------------------------------------
//| Formatted I/O Conversion Length Modifiers
//|----------------------------------------------------------------------------
//
// hh          Specifies signed char or unsigned char.
// h           Specifies short int or unsigned short int.
// l           Specifies long int or unsigned long int.
// ll          Specifies long long int or unsigned long long int.
// j           Specifies intmax_t or uintmax_t.
// z           Specifies size_t or its signed counterpart.
// t           Specifies ptrdiff_t.
// L           Specifies long double.


//|----------------------------------------------------------------------------
//| Formatted I/O Conversion Flags
//|----------------------------------------------------------------------------
//
// -           Left-justifies the converted value.
// +           Always displays a sign for signed conversions.
// space       Prefixes positive signed values with a space.
// #           Requests an alternate form of the conversion.
// 0           Pads numeric output with zeroes instead of spaces.


//|----------------------------------------------------------------------------
//| Formatted I/O Field Width and Precision
//|----------------------------------------------------------------------------
//
// number      Specifies a minimum field width.
// *           Takes the field width from an int argument.
// .number     Specifies the precision.
// .*          Takes the precision from an int argument.


//|----------------------------------------------------------------------------
//| printf Return Values
//|----------------------------------------------------------------------------
//
// printf       Returns the number of characters transmitted, or a negative value on output error.
// fprintf      Returns the number of characters transmitted, or a negative value on output error.
// sprintf      Returns the number of characters written, excluding the terminating null character.
// snprintf     Returns the number of characters that would have been written if size were sufficiently large.
