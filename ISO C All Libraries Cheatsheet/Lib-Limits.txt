//|----------------------------------------------------------------------------
//| Minimum Required Ranges
//|----------------------------------------------------------------------------

// ISO C guarantees at least the following ranges:
//
//     SCHAR_MIN   <= -127
//     SCHAR_MAX   >=  127
//     UCHAR_MAX   >=  255
//
//     SHRT_MIN    <= -32767
//     SHRT_MAX    >=  32767
//     USHRT_MAX   >= 65535
//
//     INT_MIN     <= -32767
//     INT_MAX     >=  32767
//     UINT_MAX    >= 65535
//
//     LONG_MIN    <= -2147483647
//     LONG_MAX    >=  2147483647
//     ULONG_MAX   >= 4294967295
//
//     LLONG_MIN   <= -9223372036854775807
//     LLONG_MAX   >=  9223372036854775807
//     ULLONG_MAX  >= 18446744073709551615
//
// The implementation may provide larger ranges.

//|----------------------------------------------------------------------------
//| Important char Rule
//|----------------------------------------------------------------------------

// char is a distinct type from both signed char and unsigned char.
//
// Whether plain char behaves as signed or unsigned is implementation-defined.
//
// Therefore:
//
//     CHAR_MIN
//     CHAR_MAX
//
// must be used when determining the range of plain char.
//
// Do not assume:
//
//     CHAR_MIN == SCHAR_MIN
//
// or:
//
//     CHAR_MAX == SCHAR_MAX

//|----------------------------------------------------------------------------
//| Important Width Rule
//|----------------------------------------------------------------------------

// CHAR_BIT is the number of bits in a byte.
//
// sizeof(char) is always 1.
//
// Therefore, one byte is not required by ISO C to contain exactly 8 bits.
//
// On common modern systems:
//
//     CHAR_BIT == 8
//
// but portable C code must not assume this unless the program explicitly
// requires an 8-bit byte.

//|----------------------------------------------------------------------------
//| ISO C Scope
//|----------------------------------------------------------------------------

// <limits.h> provides:
//
//     integer limits
//     character limits
//     byte size
//     multibyte character limits
//     C23 integer width information
//
// It contains no functions and no typedef declarations.
